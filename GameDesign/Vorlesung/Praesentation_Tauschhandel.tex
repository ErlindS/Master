\begin{frame}{Was ist Tauschhandel?}
    \begin{itemize}
        \item Direkter Austausch von Waren oder Leistungen ohne Geld
        \item Kein fester Preis, der Wert wird bei jedem Tausch neu ausgehandelt
        \item Lehrbuchmodell (Smith 1776, Menger 1892): Tausch als Vorstufe des
              Geldes~\cite{smith1776, menger1892}
        \item Hauptproblem: doppelte Koinzidenz der Bedürfnisse
        \item Lösung im Modell: Warengeld wie Salz, Getreide oder Vieh
    \end{itemize}
\end{frame}

\begin{frame}{Was die Forschung zeigt}
    \begin{itemize}
        \item Keine Gesellschaft belegt, die allein auf Tauschhandel
              beruhte~\cite{humphrey1985}
        \item Gemeinschaften regelten Versorgung über Geschenke und
              Schulden~\cite{graeber2011}
        \item Tauschhandel ist vor allem in zwei Fällen belegt:
        \begin{itemize}
            \item Handel mit Fremden
            \item Zusammenbruch einer Währung
        \end{itemize}
    \end{itemize}
    \vspace{0.3cm}
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            {\color{hkared}\bfseries Deutschland 1945--1948}\par\small
            Reichsmark wertlos, Zigaretten als Ersatzwährung. Ende mit der
            Währungsreform am 20.~Juni 1948.
        \end{column}
        \begin{column}{0.47\textwidth}
            {\color{hkared}\bfseries Argentinien 2002}\par\small
            Rund 4.500 Tauschclubs in allen Provinzen, etwa 2,5~Millionen
            Nutzer~\cite{ambito2002}.
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Zeitgeist: zwei Formen des Tauschens}
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            {\color{clC}\bfseries Alltag: sparen und weitergeben}
            \begin{itemize}\small
                \item Rund ein Viertel kauft aus zweiter Hand, Hauptgrund
                      Sparen~\cite{idealo2024}
                \item 6,5~Mrd.\ Euro für Secondhand 2024, 52\,\% davon
                      Kleidung~\cite{cebr2025}
                \item Über 4.000 Repair-Cafés weltweit, rund 190.000
                      Reparaturen pro Jahr~\cite{nltimes2024}
                \item Bundesförderung bis 3.000 Euro seit Dezember
                      2024~\cite{heise2024}
            \end{itemize}
        \end{column}
        \begin{column}{0.47\textwidth}
            {\color{clD}\bfseries Krise: vorsorgen und absichern}
            \begin{itemize}\small
                \item Inflation 6,9\,\% (2022) und 5,9\,\% (2023)~\cite{statistikportal}
                \item 52\,\% haben große Sorge vor steigenden Preisen
                      (2025)~\cite{ruv2025}
                \item 38\,\% hatten 2022 Lebensmittelvorräte
                      angelegt~\cite{lmpraxis2022}
                \item 29\,\% erfüllen die BBK-Empfehlung von zehn Tagen
                      Vorrat~\cite{stuttgart2024}
                \item Russland und China tauschen wegen Sanktionen Waren direkt,
                      z.\,B. Autos gegen Weizen~\cite{reuters2025}
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}
