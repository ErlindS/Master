\kopf{Die zehn Rückkopplungen der Karte}
     {Pfad, Vorzeichenprüfung, beobachtbares Verhalten und geprüfte Fundstelle}
\begin{tikzpicture}[remember picture, overlay, shift={(current page.north west)}]
\node[anchor=north west, inner sep=0pt] at (0.9,-3.85) {%
\fontsize{8.6}{10.4}\selectfont
\renewcommand{\arraystretch}{1.3}
\begin{tabularx}{32.0cm}{@{}p{2.9cm} p{9.9cm} p{0.6cm} X p{5.6cm}@{}}
\toprule
\textbf{Schleife} & \textbf{Kausalpfad} {\color{black!55}(Anzahl $-$)} & \textbf{Typ} & \textbf{Was man in der Realität beobachtet} & \textbf{Fundstelle} \\
\midrule
\textbf{B1} Preisausgleich &
  Angebot $\xrightarrow{-}$ Preis $\xrightarrow{-}$ Kaufrate $\xrightarrow{-}$ Angebot \hfill (3) & B &
  Wird Ware knapp, steigt der Preis, es wird weniger gekauft, das Angebot erholt sich. &
  Sterman Kap.~5.5, S.~169 \\
\textbf{B2} Nachschub &
  Preis $\xrightarrow{+}$ Marge $\xrightarrow{+}$ Ware unterwegs $\xrightarrow{+\,\|}$ Angebot $\xrightarrow{-}$ Preis \hfill (1) & B &
  Nachschub braucht Zeit, also wird zu lange zu viel bestellt. Preise schwingen über das Gleichgewicht hinaus und zurück. &
  Sterman Kap.~17, S.~663; Kap.~20, S.~791; Meadows S.~51--58 \\
\textbf{R1} Spekulation &
  Preis $\xrightarrow{+}$ spekul.\ Nachfrage $\xrightarrow{+}$ Kaufrate $\xrightarrow{-}$ Angebot $\xrightarrow{-}$ Preis \hfill (2) & R &
  Steigende Preise locken Käufer an, die auf weitere Steigerung setzen. Die Blase wächst, bis B1 sie bremst. &
  Sterman Kap.~5.5, S.~173 \\
\textbf{R3} Vertrauen &
  Vertrauen $\xrightarrow{-}$ Transaktionskosten $\xrightarrow{-}$ Kaufrate $\xrightarrow{+}$ Vertrauen \hfill (2) & R &
  Wer verlässlich liefert, wird ohne Prüfung gekauft. Jedes gute Geschäft stärkt den Ruf; ein Skandal dreht die Schleife um. &
  Sterman Kap.~18.2.1, S.~735; Salen/Zimmerman Kap.~19, S.~230 \\
\textbf{B4} Opportunismus &
  Vertrauen $\xrightarrow{+}$ Betrug $\xrightarrow{-}$ Vertrauen \hfill (1) & B &
  Wo viel vertraut wird, lohnt sich Betrug (Fake-Shops). Aufgeflogene Fälle senken das Vertrauen. Aufsicht dämpft den Betrug. &
  Salen/Zimmerman Kap.~21, S.~266 \\
\textbf{B3} Budget &
  Kaufrate $\xrightarrow{-\,\|}$ Budget $\xrightarrow{+}$ Kaufrate \hfill (1) & B &
  Kaufen leert das Konto. Ratenkauf verzögert diese Bremse, deshalb wird die Grenze oft erst spät bemerkt. &
  Sterman Kap.~11, S.~409; Adams/Dormans Kap.~4, S.~59 \\
\textbf{R2} Daten &
  Kaufrate $\xrightarrow{+}$ Daten $\xrightarrow{+}$ gezielte Kaufanreize $\xrightarrow{+}$ Kaufrate \hfill (0) & R &
  Mehr Käufe liefern mehr Daten. Damit werden Angebote, Preise und Countdowns passgenauer. &
  Sterman Kap.~10.4, S.~364--370; Salen/Zimmerman Kap.~17, S.~202 \\
\textbf{R4} Erfolg dem\newline Erfolgreichen &
  Marktmacht $\xrightarrow{+}$ Marge $\xrightarrow{+}$ Kapital $\xrightarrow{+}$ Marktmacht \hfill (0) & R &
  Wer groß ist, diktiert Einkaufskonditionen und Gebühren und wird dadurch noch größer. &
  Meadows S.~126--130; Sterman Kap.~10.4.7, S.~374 \\
\textbf{B5} Konkurrenz &
  Marge $\xrightarrow{+\,\|}$ Konkurrenz $\xrightarrow{-}$ Marge \hfill (1) & B &
  Hohe Margen locken Nachahmer an, die Preise und Margen drücken. Wirkt verzögert und oft schwächer als R4. &
  Adams/Dormans Anh.~B, S.~316 (Dynamic Friction) \\
\textbf{B6} Gegenbewegung &
  Marktmacht $\xrightarrow{+}$ öffentl.\ Druck $\xrightarrow{+\,\|}$ Regulierung $\xrightarrow{-}$ Marktmacht \hfill (1) & B &
  Missbrauch wird sichtbar, Gesetze folgen mit Jahren Verzug (LkSG 2021, CSDDD 2024). Betroffene suchen Umgehungen. &
  Meadows S.~112 (Policy Resistance), S.~136 (Rule Beating); Salen/Zimmerman Kap.~18, S.~212 \\
\bottomrule
\end{tabularx}};

\node[anchor=north west, text width=31.8cm, font=\footnotesize, align=left] at (0.9,-16.3) {%
  \textbf{Prüfregel (Sterman Kap.~5.2.3, S.~143):} Eine Schleife ist verstärkend (R), wenn die Zahl ihrer negativen Kanten gerade ist, sonst ausgleichend (B).\\[3pt]
  \textbf{Warum eine Karte statt zehn Einzelbilder?} Die Schleifen teilen sich Größen. Ob ein Preisanstieg durch B1 gebremst oder durch R1 verstärkt wird,
  entscheidet die \emph{Stärke} der Schleifen. Das sieht man nur, wenn sie in einem Bild verbunden sind (Meadows S.~126). Sterman warnt zugleich vor zu großen
  Diagrammen (Kap.~5.2.9, S.~154), daher die Beschränkung auf zehn Schleifen.};
\end{tikzpicture}
