\begin{frame}{Was wurde gemacht: Lastskripte (k6)}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Messlauf \texttt{eshop.js}}
      \begin{itemize}\small
        \item feste Ankunftsrate (offenes Modell), Standard 10 Sessions/s
        \item Mix: 80\,\% Stöbern, 20\,\% Bestellen (deterministisch)
        \item Phasen: 2\,min Aufwärmen, 5\,min Messung, 1\,min Ende
        \item Parameter per \texttt{-e}: \texttt{RATE}, \texttt{RUN\_ID}, \texttt{ORDER\_SHARE}, \dots
        \item Zusammenfassung je Lauf als JSON
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Ablauf einer Session}
      \begin{itemize}\small
        \item \emph{Stöbern:} Startseite, Artikelliste, Filter, 1--3 Detailseiten
        \item \emph{Bestellen:} Login, Warenkorb (gRPC), \texttt{POST /api/orders}, Bestellliste
        \item danach asynchron über RabbitMQ: OrderProcessor, PaymentProcessor, Lagerbestand
      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{0.4cm}
  \textbf{Gültigkeit eines Laufs} {\small(nur Messphase, sonst Exit-Code 99)}
  \begin{itemize}\small
    \item HTTP-Fehler $\le$ 1\,\%, Checks und Bestellungen $>$ 99\,\% erfolgreich
    \item keine verworfenen Iterationen (Zielrate erreicht), p95 $<$ 2\,s
  \end{itemize}
\end{frame}
